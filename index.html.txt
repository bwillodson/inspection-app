        function setup(id) {
            const c = document.getElementById(id); const x = c.getContext("2d");
            c.width = c.offsetWidth; c.height = c.offsetHeight;
            let d = false;
            const p = (e) => { const r = c.getBoundingClientRect(); return { x: (e.touches ? e.touches[0].clientX : e.clientX) - r.left, y: (e.touches ? e.touches[0].clientY : e.clientY) - r.top }; };
            c.addEventListener("touchstart", (e) => { e.preventDefault(); d = true; x.beginPath(); x.moveTo(p(e).x, p(e).y); });
            c.addEventListener("touchmove", (e) => { e.preventDefault(); if(d) { x.lineTo(p(e).x, p(e).y); x.stroke(); } });
            window.addEventListener("touchend", () => d = false);
            c.addEventListener("mousedown", (e) => { d = true; x.beginPath(); x.moveTo(p(e).x, p(e).y); });
            c.addEventListener("mousemove", (e) => { if(d) { x.lineTo(p(e).x, p(e).y); x.stroke(); } });
        }
        setup("c_insp"); setup("c_cli");
